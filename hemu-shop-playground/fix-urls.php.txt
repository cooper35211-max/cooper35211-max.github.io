<?php
// Playground 匯入後執行：把資料庫裡本機站的網址（http://localhost:8080）換成 Playground 的網址。
// 序列化的設定值先反序列化再取代，避免長度不符把設定弄壞。
require '/wordpress/wp-load.php';
global $wpdb;
$old = 'http://localhost:8080';
$new = untrailingslashit( home_url() );
$swap = function ( $v ) use ( &$swap, $old, $new ) {
	if ( is_string( $v ) ) { return str_replace( $old, $new, $v ); }
	if ( is_array( $v ) ) { foreach ( $v as $k => $x ) { $v[ $k ] = $swap( $x ); } return $v; }
	if ( is_object( $v ) ) { foreach ( get_object_vars( $v ) as $k => $x ) { $v->$k = $swap( $x ); } return $v; }
	return $v;
};
$n = 0;
foreach ( $wpdb->get_results( "SELECT ID, post_content, post_excerpt, guid FROM {$wpdb->posts} WHERE post_content LIKE '%localhost:8080%' OR post_excerpt LIKE '%localhost:8080%' OR guid LIKE '%localhost:8080%'" ) as $p ) {
	$wpdb->update( $wpdb->posts, array( 'post_content' => $swap( $p->post_content ), 'post_excerpt' => $swap( $p->post_excerpt ), 'guid' => $swap( $p->guid ) ), array( 'ID' => $p->ID ) );
	$n++;
}
foreach ( $wpdb->get_results( "SELECT option_name FROM {$wpdb->options} WHERE option_value LIKE '%localhost:8080%' AND option_name NOT IN ('siteurl','home')" ) as $o ) {
	update_option( $o->option_name, $swap( get_option( $o->option_name ) ) );
	$n++;
}
foreach ( $wpdb->get_results( "SELECT meta_id, meta_value FROM {$wpdb->postmeta} WHERE meta_value LIKE '%localhost:8080%'" ) as $m ) {
	$wpdb->update( $wpdb->postmeta, array( 'meta_value' => maybe_serialize( $swap( maybe_unserialize( $m->meta_value ) ) ) ), array( 'meta_id' => $m->meta_id ) );
	$n++;
}

// Playground 限定：綠界的付款頁與超商門市地圖不允許放在別人的框架裡，Playground 又是把網站放在框架中執行，
// 所以加一個 WooCommerce 內建的「銀行轉帳」當展示用付款，訪客才能完整走完下單、在後台看到訂單。綠界選項照樣保留。
update_option( 'woocommerce_bacs_settings', array(
	'enabled'      => 'yes',
	'title'        => '展示用付款（Playground 限定，不會扣款）',
	'description'  => '這是作品集展示站。綠界付款頁無法在 Playground 中開啟，選這個可以直接完成訂單、到後台查看。',
	'instructions' => '展示用訂單，不需要付款。',
	'account_details' => '',
) );
update_option( 'permalink_structure', '/%postname%/' );
update_option( 'woocommerce_coming_soon', 'no' );
flush_rewrite_rules();
wp_cache_flush();
echo "replaced $n rows -> $new\n";
